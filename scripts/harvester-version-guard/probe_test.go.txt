type versions struct{ AdapterVersion, IndexerVersion int }
type guardSnapshot struct {
	Versions map[string]versions
	Outputs  map[string]json.RawMessage
}

// Only the runner injects this test into temporary archived revisions.
func TestHarvesterGuardCapture(t *testing.T) {
	corpus := os.Getenv("PEASANT_GUARD_CORPUS")
	result := guardSnapshot{Versions: parserVersions(), Outputs: make(map[string]json.RawMessage)}
	read := func(path string, target any) bool {
		data, err := os.ReadFile(filepath.Join(corpus, path))
		if os.IsNotExist(err) {
			return false
		} // Corpus may predate this fixture family.
		if err != nil {
			t.Fatal(err)
		}
		if err := yaml.Unmarshal(data, target); err != nil {
			t.Fatal(err)
		}
		return true
	}
	write := func(filesystem ingest.FileSystem, path, data string) {
		if err := filesystem.WriteFile(path, []byte(data), 0600); err != nil {
			t.Fatal(err)
		}
	}
	observe := func(name string, session ingest.DiscoveredSession, filesystem ingest.FileSystem) {
		session.ModTime = time.UnixMilli(1700000000000)
		session.CreatedAt = session.ModTime
		if session.TranscriptOrigin == ingest.TranscriptOriginFile {
			adapter := ingest.DefaultAdapterRegistry[session.Harness](filesystem, testutil.DefaultGitResolver(), salt.Salt{})
			metadata, adapterErr := adapter.ExtractMetadata(t.Context(), session)
			guardRecord(t, &result, string(session.Harness)+"/AdapterVersion/"+name, guardMetadata(t, metadata, ""), adapterErr)
		}
		indexer := ingest.NewIndexerRegistry(filesystem, ingest.IndexerRegistryOptions{})[session.Harness]
		var output any
		var indexErr error
		// The old API returned the V1 entry slice directly. The new API's
		// completion-checked method is the production persistence boundary.
		method := reflect.ValueOf(indexer).MethodByName("IndexTranscriptResult")
		if method.IsValid() {
			values := method.Call([]reflect.Value{reflect.ValueOf(t.Context()), reflect.ValueOf(session)})
			if !values[1].IsNil() {
				indexErr = values[1].Interface().(error)
			}
			if !values[0].IsNil() {
				concrete := values[0].Interface()
				format := concrete.(interface{ IndexVersion() int }).IndexVersion()
				output = map[string]any{"format": format, "result": concrete}
			}
		} else {
			entries, err := indexer.IndexTranscript(t.Context(), session)
			indexErr = err
			if err == nil {
				output = map[string]any{"format": 1, "result": map[string]any{"Entries": entries}}
			}
		}
		guardRecord(t, &result, string(session.Harness)+"/IndexerVersion/"+name, output, indexErr)
	}

	// Existing native E2E corpora exercise full adapter extraction and shared
	// tool/entry helpers for Claude, Codex and Cursor; filenames are input IDs.
	for _, harness := range []string{"claude", "codex", "cursor"} {
		root := "internal/e2e/testdata/" + harness + "-fixture"
		var manifest struct {
			Harness  ingest.Harness `yaml:"harness"`
			Sessions []struct {
				ID   string `yaml:"id"`
				Path string `yaml:"path"`
			} `yaml:"sessions"`
		}
		if !read(root+"/fixture-index.yaml", &manifest) {
			t.Fatalf("required native corpus missing: %s", root)
		}
		for _, row := range manifest.Sessions {
			filesystem := testutil.NewMemFS()
			data, err := os.ReadFile(filepath.Join(corpus, root, row.Path))
			if err != nil {
				t.Fatal(err)
			}
			path := "/synthetic/" + row.Path
			write(filesystem, path, string(data))
			id, err := ingest.NewSessionID(row.ID)
			if err != nil {
				t.Fatal(err)
			}
			observe("native/"+row.Path, ingest.DiscoveredSession{SessionID: id, Harness: manifest.Harness, SourcePath: ingest.ResolvedPath(path), SourceFormat: ingest.SourceFormatJSONL}, filesystem)
		}
	}
	filesystem := testutil.NewMemFS()
	for _, suffix := range []string{"jsonl", "meta.json"} {
		data, err := os.ReadFile(filepath.Join(corpus, "internal/ingest/testdata/strike_protocol."+suffix))
		if err != nil {
			t.Fatal(err)
		}
		write(filesystem, "/synthetic/strike_protocol."+suffix, string(data))
	}
	observe("strike_protocol", ingest.DiscoveredSession{SessionID: testutil.TestSessionUUID, Harness: ingest.HarnessStrike, SourcePath: "/synthetic/strike_protocol.jsonl", SourceFormat: ingest.SourceFormatJSONL}, filesystem)

	// Reuse the named completion cases, including old failure/new success and
	// successful empty versus failed parses. Expectations are never digests.
	var completion struct {
		RequiredNames []string `yaml:"requiredNames"`
		Cases         []struct {
			Name        string         `yaml:"name"`
			Harness     ingest.Harness `yaml:"harness"`
			Input       string         `yaml:"input"`
			Missing     bool           `yaml:"missing"`
			LongContent bool           `yaml:"longContent"`
			Oversized   bool           `yaml:"oversized"`
			PartialRead bool           `yaml:"partialRead"`
		} `yaml:"cases"`
	}
	if read("internal/ingest/testdata/indexer_completion.yaml", &completion) {
		seen := make(map[string]bool)
		for _, row := range completion.Cases {
			guardName(t, seen, row.Name)
			filesystem := &guardFaultFS{MemFS: testutil.NewMemFS(), partial: row.PartialRead}
			data := row.Input
			if row.LongContent {
				data = strings.ReplaceAll(data, "LONG_CONTENT", strings.Repeat("x", 2050))
			}
			if row.Oversized {
				data += strings.Repeat("x", defaults.ScannerMaxLine) + "\n"
			}
			if !row.Missing {
				write(filesystem, "/synthetic/transcript.jsonl", data)
			}
			observe(row.Name, ingest.DiscoveredSession{SessionID: testutil.TestSessionUUID, Harness: row.Harness, SourcePath: "/synthetic/transcript.jsonl", SourceFormat: ingest.SourceFormatJSONL}, filesystem)
		}
		for _, name := range completion.RequiredNames {
			if !seen[name] {
				t.Fatalf("required completion case missing: %s", name)
			}
		}
	}
	var openCode struct {
		RequiredNames []string `yaml:"requiredNames"`
		Cases         []struct {
			Name        string            `yaml:"name"`
			Files       map[string]string `yaml:"files"`
			Directories []string          `yaml:"directories"`
			ReadError   string            `yaml:"readError"`
			DirError    string            `yaml:"dirError"`
			Managed     string            `yaml:"managed"`
			Origin      string            `yaml:"origin"`
		} `yaml:"cases"`
	}
	if read("internal/ingest/testdata/opencode_index_completion.yaml", &openCode) {
		seen := make(map[string]bool)
		for _, row := range openCode.Cases {
			guardName(t, seen, row.Name)
			filesystem := &guardFaultFS{MemFS: testutil.NewMemFS()}
			session := setupOpenCodeFixture(t, filesystem.MemFS, testutil.TestOpenCodeSesID, "synthetic")
			expand := strings.NewReplacer("SESSION_ID", session.SessionID.String(), "LONG_CONTENT", strings.Repeat("x", 2050))
			for path, data := range row.Files {
				write(filesystem, filepath.Join("/opencode-store/storage", expand.Replace(path)), expand.Replace(data))
			}
			for _, path := range row.Directories {
				if err := filesystem.MkdirAll(filepath.Join("/opencode-store/storage", expand.Replace(path)), 0700); err != nil {
					t.Fatal(err)
				}
			}
			if row.ReadError != "" {
				filesystem.readError = filepath.Join("/opencode-store/storage", expand.Replace(row.ReadError))
			}
			if row.DirError != "" {
				filesystem.dirError = filepath.Join("/opencode-store/storage", expand.Replace(row.DirError))
			}
			if row.Origin != "" {
				// The indexer receives fixed managed bytes, independently of the
				// adapter's output, so an adapter change alone does not blame it.
				session.TranscriptOrigin = ingest.TranscriptOriginOpenCodeCurrentSQLite
				if row.Origin == "legacy" {
					session.TranscriptOrigin = ingest.TranscriptOriginOpenCodeLegacySQLite
				}
				session.SourcePath = "/synthetic/projection.json"
				write(filesystem, session.SourcePath.String(), expand.Replace(row.Managed))
			}
			observe(row.Name, session, filesystem)
		}
		for _, name := range openCode.RequiredNames {
			if !seen[name] {
				t.Fatalf("required OpenCode case missing: %s", name)
			}
		}
	} else {
		// This native shape and helper predate the completion fixture family.
		filesystem := testutil.NewMemFS()
		session := setupOpenCodeFixture(t, filesystem, testutil.TestOpenCodeSesID, "synthetic")
		observe("native-directory", session, filesystem)
	}
	// Reuse native SQLite shapes from the existing materialization corpus.
	// The runner installs the same test-only builder and YAML for both revisions.
	nativeNames := map[string]bool{"legacy-message-part": true, "current-session-message": true}
	var native struct {
		Cases []struct {
			SourceFixture string `yaml:"source_fixture"`
		} `yaml:"cases"`
	}
	if read("internal/ingest/testdata/opencode_native_rows.yaml", &native) {
		for _, row := range native.Cases {
			nativeNames[row.SourceFixture] = true
		}
	}
	for name := range nativeNames {
		source := testfixture.MaterializeByName(t, name)
		// Reuse the exact same physical locator and file clock across runs:
		// unattributed sessions derive project identity from this path.
		nativePath := filepath.Join(os.Getenv("PEASANT_GUARD_NATIVE"), name, "opencode.db")
		if err := os.MkdirAll(filepath.Dir(nativePath), 0700); err != nil {
			t.Fatal(err)
		}
		nativeBytes, err := os.ReadFile(source.Path)
		if err != nil {
			t.Fatal(err)
		}
		if err := os.WriteFile(nativePath, nativeBytes, 0600); err != nil {
			t.Fatal(err)
		}
		clock := time.UnixMilli(1700000000000)
		if err := os.Chtimes(nativePath, clock, clock); err != nil {
			t.Fatal(err)
		}
		filesystem := &ingest.OSFileSystem{}
		adapter, err := ingest.NewOpenCodeAdapterWithCandidateProbe(filesystem, testutil.DefaultGitResolver(), salt.Salt{}, "latest", guardEnvironment{"OPENCODE_DB": nativePath}, filesystem, ingest.OpenOpenCodeSQLiteSource, ingest.DefaultOpenCodeSQLiteSourceOptions())
		if err != nil {
			t.Fatal(err)
		}
		root := ingest.ResolvedPath(filepath.Dir(nativePath))
		sessions, discoverErr := adapter.Discover(t.Context(), ingest.SourceConfig{Enabled: true, Paths: []ingest.ResolvedPath{root}})
		outputs := make(map[string]any)
		for _, session := range sessions {
			metadata, data, materializeErr := adapter.MaterializeTranscript(t.Context(), session)
			var transcript any
			if len(data) > 0 {
				if err := json.Unmarshal(data, &transcript); err != nil {
					t.Fatal(err)
				}
			}
			outputs[session.SessionID.String()] = struct {
				Failed               bool
				Metadata, Transcript any
			}{materializeErr != nil, guardMetadata(t, metadata, filepath.Dir(nativePath)), transcript}
		}
		guardRecord(t, &result, "opencode/AdapterVersion/native/"+name, outputs, discoverErr)
	}
	for harness := range result.Versions {
		for _, stage := range []string{"AdapterVersion", "IndexerVersion"} {
			found := false
			for name := range result.Outputs {
				found = found || strings.HasPrefix(name, harness+"/"+stage+"/")
			}
			if !found {
				t.Fatalf("registered harness %s has no %s corpus; add native fixture coverage", harness, stage)
			}
		}
	}
	data, err := json.Marshal(result)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(os.Getenv("PEASANT_GUARD_OUTPUT"), data, 0600); err != nil {
		t.Fatal(err)
	}
}

func guardName(t *testing.T, seen map[string]bool, name string) {
	t.Helper()
	if name == "" || seen[name] {
		t.Fatalf("missing or duplicate corpus name %q", name)
	}
	seen[name] = true
}

func guardMetadata(t *testing.T, metadata *ingest.UnifiedMetadata, temporaryRoot string) any {
	t.Helper()
	if metadata == nil {
		return nil
	}
	data, err := json.Marshal(metadata)
	if err != nil {
		t.Fatal(err)
	}
	if temporaryRoot != "" {
		data = []byte(strings.ReplaceAll(string(data), temporaryRoot, "/synthetic/native"))
	}
	var object map[string]any
	if err := json.Unmarshal(data, &object); err != nil {
		t.Fatal(err)
	}
	// Serialization/producer bookkeeping and hashes derived from it are not
	// adapter semantics. Keep all actual extracted timestamps, stats and context.
	for _, key := range []string{"schemaVersion", "adapterVersion", "contentHash", "metadataHash", "artifactHash", "derivedAt"} {
		delete(object, key)
	}
	if timestamp, ok := object["timestamp"].(map[string]any); ok {
		delete(timestamp, "ingested")
	}
	return object
}

type guardEnvironment map[string]string

func (e guardEnvironment) LookupEnv(key string) (string, bool) { value, ok := e[key]; return value, ok }

func guardRecord(t *testing.T, result *guardSnapshot, name string, value any, err error) {
	t.Helper()
	if _, exists := result.Outputs[name]; exists {
		t.Fatalf("duplicate observation %s", name)
	}
	// Diagnostic wording does not alter persisted parser behavior; failure
	// versus success does. Preserve partial output when a parser returns it.
	data, marshalErr := json.Marshal(struct {
		Failed bool
		Output any
	}{err != nil, value})
	if marshalErr != nil {
		t.Fatal(marshalErr)
	}
	result.Outputs[name] = data
}

type guardFaultFS struct {
	*testutil.MemFS
	partial             bool
	readError, dirError string
}

var _ ingest.FileSystem = (*guardFaultFS)(nil)

func (f *guardFaultFS) ReadFile(path string) ([]byte, error) {
	if path == f.readError {
		return nil, &fs.PathError{Op: "read", Path: path, Err: fs.ErrPermission}
	}
	data, err := f.MemFS.ReadFile(path)
	if err == nil && f.partial {
		err = fmt.Errorf("synthetic partial source read")
	}
	return data, err
}

func (f *guardFaultFS) ReadDir(path string) ([]os.DirEntry, error) {
	if path == f.dirError {
		return nil, &fs.PathError{Op: "readdir", Path: path, Err: fs.ErrPermission}
	}
	return f.MemFS.ReadDir(path)
}
